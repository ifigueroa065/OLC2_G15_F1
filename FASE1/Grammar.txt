Program
  = _ instructions:InstructionList? _ { return { type: "Program", body: instructions || [] }; }

InstructionList
  = head:Instruction tail:(_ (EOL / ";") _ Instruction)* {
      return [head, ...tail.map(e => e[3])].filter(instr => instr !== null);
    }

Instruction
  = VariableAssignment 
  / Directive
  / LoadInstruction
  / StoreInstruction
  / ArithmeticInstruction
  / LogicalInstruction
  / ShiftRotateInstruction
  / BranchInstruction
  / ConditionalBranchInstruction
  / UnconditionalBranchInstruction
  / RelationalOperatorInstruction
  / MoveInstruction
  / SVCInstruction
  / Label
  / LineComment
  / SemicolonComment
  / BlockComment
  
Directive
  = "." directive:(reserv) _ value:(reserv) { return { type: "Directive", name, value }; }
  / "." directive:(reserv) {return {type: "Directive", name};}

VariableAssignment
  = variable:Identifier ":" _ "." directive:(reserv) _ value:(Number / StringLiteral) {
      return { type: "VariableAssignment", variable, directive, value };
    }

LoadInstruction
  = op:("ldr"i / "ldrb"i / "ldp"i) _ dest:Register "," _ value:(Address / Literal) {
      return { type: "LoadInstruction", op, dest, value };
    }

StoreInstruction
  = op:("strb"i / "str"i / "stp"i) _ src:Register "," _ "[" dest:Register "]" {
      return { type: "StoreInstruction", op, src, dest };
    }

ArithmeticInstruction
  = op:("add"i / "sub"i / "mul"i / "udiv"i / "sdiv"i) _ dest:Register "," _ src1:Register "," _ src2:(Register / ImmediateValue) {
      return { type: "ArithmeticInstruction", op, dest, src1, src2 };
    }

LogicalInstruction
  = op:("and"i / "orr"i / "eor"i / "mvn"i) _ dest:Register "," _ src1:Register ("," _ src2:Register)? {
      return { type: "LogicalInstruction", op, dest, src1, src2: src2 || null };
    }

ShiftRotateInstruction
  = op:("lsl"i / "lsr"i / "asr"i / "ror"i) _ dest:Register "," _ src:Register "," _ "#" value:Number {
      return { type: "ShiftRotateInstruction", op, dest, src, value };
    }

BranchInstruction
  = op:( "ret"i / "beq"i / "bne"i / "bgt"i / "blt"i /"bl"i / "b"i ) _ target:Identifier {
      return { type: "BranchInstruction", op, target };
    }

ConditionalBranchInstruction
  = "b." condition:ConditionCode _ label:Identifier {
      return { type: "ConditionalBranchInstruction", condition, label };
    }

UnconditionalBranchInstruction
  = "b" _ label:Identifier {
      return { type: "UnconditionalBranchInstruction", label };
    }

RelationalOperatorInstruction
  = "cmp" _ reg1:Register comma _ reg2:(Register / ImmediateValue) {
      return { type: "RelationalOperatorInstruction", reg1, reg2 };
    }

MoveInstruction
  = "mov"i _ dest:Register comma _ value:(ImmediateValue / Number / CharLiteral) {
      return { type: "MoveInstruction", dest, value: value.num !== undefined ? value.num : (value.char !== undefined ? value.char.charCodeAt(0) : value) };
    }

SVCInstruction
  = "svc"i _ value:Number { 
      return { type: "SVCInstruction", value }; 
    }

ConditionCode
  = "eq"i { return "eq"; }
  / "ne"i { return "ne"; }
  / "gt"i { return "gt"; }
  / "lt"i { return "lt"; }
  
reserv = "data"
  / "section"
  / "global"
  / "_start"
  / "text"
  / "space"
  / "word"
  / "byte"
  / "half"
  / "ascii"
  / "asciz"
  / "float"
  / "double"
  / "bss"

comma 
	= "," {return ",";}

Label
  = name:Identifier ":" { return { type: "Label", name }; }

LineComment
  = "//" comment:$([^"\n"]*) { return null; }

SemicolonComment
  = ";" comment:$([^"\n"]*) { return null; }

BlockComment
  = "/*" comment:$((!"*/" .)*) "*/" { return null; }

Register
  = ("x" / "w") [0-9]+ { return { type: "Register", name: text() }; }

Identifier
  = [a-zA-Z_][a-zA-Z0-9_]* { return text(); }

Number
  = [0-9]+ { return parseInt(text(), 10); }

Address
  = "[" base:Register "]" { return { type: "Address", base }; }

Literal
  = "=" identifier:Identifier { return identifier; }

ImmediateValue
  = "#" num:Number { return { num }; }

CharLiteral
  = "#" "'" char:[^'] "'" { return { char }; }

StringLiteral
  = '"' chars:([^'"']*) '"' { return chars.join(''); }

EOL "end of line"
  = ("\n" / "\r\n")*

_ "whitespace"
  = ( [ \t\r\n] / LineComment / SemicolonComment / BlockComment )*
